// Submit Payment, Save to Revenue Ledger, & Generate Receipt document.getElementById('feeForm').addEventListener('submit', function(e) { e.preventDefault(); const id = document.getElementById('studentId').value.trim(); const month = document.getElementById('month').value; const amount = parseFloat(document.getElementById('amount').value); if (!id || !loadedStudent) return alert('⚠️ Please lookup student details first.'); const now = new Date(); const yearMonth = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`; // e.g., '2026-09' const dateString = now.toISOString().split('T')[0]; const recNo = '#REC-' + Math.floor(1000 + Math.random() * 9000); // 1. Update Student Profile Status const studentUpdate = { fee: 'Paid', lastPaymentMonth: month, lastPaymentAmount: amount, lastPaymentDate: dateString }; db.ref('students/' + id).update(studentUpdate).then(() => { // 2. Transaction Record for Revenue Ledger const paymentRecord = { studentId: id, studentName: loadedStudent.name || 'N/A', grade: loadedStudent.grade || 'N/A', month: month, amount: amount, date: dateString, timestamp: now.toISOString(), receiptNo: recNo }; // Write to monthly ledger db.ref(`revenue/${yearMonth}/transactions/${recNo.replace('#', '')}`).set(paymentRecord); // 3. Atomically update Total Monthly Revenue Counter for Teacher Dashboard db.ref(`revenue/${yearMonth}/totalAmount`).transaction((currentTotal) => { return (currentTotal || 0) + amount; }); // 4. Update Grade-wise Revenue Breakdown const gradeKey = (loadedStudent.grade || 'Unknown').replace(/\s+/g, '_'); db.ref(`revenue/${yearMonth}/byGrade/${gradeKey}`).transaction((currentGradeTotal) => { return (currentGradeTotal || 0) + amount; }); alert('✅ Payment recorded! Revenue dashboard updated.'); // (Receipt generation & WhatsApp trigger script continues below...) generateReceiptAndOpenWhatsApp(recNo, month, amount); }).catch(err => alert('❌ Error saving payment: ' + err.message)); });
