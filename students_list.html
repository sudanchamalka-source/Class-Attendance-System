// Place this inside students_list.html to load and listen for payment updates function loadAllStudents() { const tableBody = document.getElementById('studentsTableBody'); // Real-time listener: Updates table instantly when a fee is paid db.ref('students').on('value', (snapshot) => { tableBody.innerHTML = ''; // Clear table if (!snapshot.exists()) { tableBody.innerHTML = 'No students found.'; return; } snapshot.forEach((childSnap) => { const stu = childSnap.val(); const id = childSnap.key; // Retrieve phone number variations const phone = stu.phone || stu.whatsapp || stu.parentPhone || 'N/A'; // Check fee status const feeStatus = stu.fee === 'Paid' || stu.status === 'Paid' ? `Paid (${stu.lastPaymentMonth || 'Yes'})` : `Pending`; const row = ` ${id} ${stu.name || 'N/A'} ${stu.grade || 'N/A'} ${phone} ${feeStatus} ${stu.lastPaymentAmount ? 'LKR ' + stu.lastPaymentAmount : '-'} `; tableBody.innerHTML += row; }); }); } // Call on page load window.onload = loadAllStudents;
